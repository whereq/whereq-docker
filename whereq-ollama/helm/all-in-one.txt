values.yaml
# -- Image
image:
  repository: registry.kube.dp.ist.bns/dp-hermes
  tag: v1.0.0
  pullPolicy: IfNotPresent

imagePullSecrets: []
nameOverride: ""
fullnameOverride: "hermes"

replicaCount: 1

serviceAccount:
  create: true
  name: ""

# -- Hermes runtime config
hermes:
  # Gateway / dashboard bind address inside the pod
  gatewayEndpoint: "0.0.0.0:8642"
  dashboardPort: 8642
  logLevel: "INFO"

  # LLM provider: in-cluster Ollama
  llm:
    provider: "openai"   # Ollama exposes an OpenAI-compatible API at /v1
    baseUrl: "http://ollama.ollama.svc.cluster.local:11434/v1"
    model: "qwen3:8b"    # change to whichever model you've pulled in ollama
    # Ollama doesn't require a real key, but the OpenAI client lib insists on one
    apiKey: "ollama"

  # Messaging providers — all disabled at first
  providers:
    slack:    { enabled: false }
    discord:  { enabled: false }
    telegram: { enabled: false }
    matrix:   { enabled: false }

# -- Service for the dashboard UI
service:
  type: ClusterIP
  port: 8642
  # set to NodePort or LoadBalancer if you don't want to use the ingress
  # nodePort: 30864

# -- Ingress for the dashboard
ingress:
  enabled: true
  className: "nginx"
  host: hermes.kube.dp.dev.bns
  path: /
  pathType: Prefix
  tls:
    enabled: true
    secretName: hermes-tls
  annotations:
    nginx.ingress.kubernetes.io/backend-protocol: "HTTP"
    nginx.ingress.kubernetes.io/proxy-body-size: "20m"

resources:
  requests:
    cpu: "500m"
    memory: "1Gi"
  limits:
    cpu: "2"
    memory: "4Gi"

nodeSelector: {}
tolerations: []
affinity: {}

podSecurityContext:
  fsGroup: 1000
securityContext: {}

# Pod-level extras
podAnnotations: {}
terminationGracePeriodSeconds: 30

# Probe tuning — defaults are tuned to survive the slow s6 chown of /opt/hermes
probes:
  startup:
    periodSeconds: 10
    failureThreshold: 90      # up to ~15 minutes on first boot
    timeoutSeconds: 3
  readiness:
    periodSeconds: 10
    failureThreshold: 3
    timeoutSeconds: 3
  liveness:
    periodSeconds: 30
    failureThreshold: 5
    timeoutSeconds: 3

# Dynatrace OneAgent injection. The LD_PRELOAD wrapper roughly doubles syscall
# latency, which makes the recursive chown during s6 init painfully slow.
# Disable here unless platform team requires APM on this workload.
dynatrace:
  enabled: false

# -- Persistence for chat history / vector store / cache
persistence:
  enabled: true
  storageClass: "longhorn"   # change to a storageclass you saw in `kubectl get sc`
  accessModes: ["ReadWriteOnce"]
  size: 10Gi
  mountPath: /app/data
  
  fixPermissions: true
  fixPermissionsImage: "busybox:1.36"
  
# Optional escape hatches for extra mounts (e.g. CA bundles, extra configs)
extraVolumes: []
extraVolumeMounts: []

deployment.yaml
apiVersion: apps/v1
kind: Deployment
metadata:
  name: {{ include "hermes.fullname" . }}
  labels:
    {{- include "hermes.labels" . | nindent 4 }}
spec:
  replicas: {{ .Values.replicaCount }}
  selector:
    matchLabels:
      {{- include "hermes.selectorLabels" . | nindent 6 }}
  template:
    metadata:
      labels:
        {{- include "hermes.selectorLabels" . | nindent 8 }}
      annotations:
        # Roll the pods if config/secret content changes
        checksum/config: {{ include (print $.Template.BasePath "/configmap.yaml") . | sha256sum }}
        checksum/secret: {{ include (print $.Template.BasePath "/secret.yaml") . | sha256sum }}
        {{- if not .Values.dynatrace.enabled }}
        # Opt out of Dynatrace OneAgent injection (LD_PRELOAD wrapper slows down s6 init)
        dynakube.dynatrace.com/inject: "false"
        oneagent.dynatrace.com/inject: "false"
        metadata-enrichment.dynatrace.com/inject: "false"
        {{- end }}
        {{- with .Values.podAnnotations }}
        {{- toYaml . | nindent 8 }}
        {{- end }}
    spec:
      serviceAccountName: {{ include "hermes.serviceAccountName" . }}
      {{- with .Values.imagePullSecrets }}
      imagePullSecrets:
        {{- toYaml . | nindent 8 }}
      {{- end }}
      securityContext:
        {{- toYaml .Values.podSecurityContext | nindent 8 }}
      {{- if and .Values.persistence.enabled .Values.persistence.fixPermissions }}
      initContainers:
        - name: fix-data-perms
          image: {{ .Values.persistence.fixPermissionsImage | default "busybox:1.36" }}
          command:
            - sh
            - -c
            - |
              set -e
              echo "Fixing ownership of {{ .Values.persistence.mountPath }} to 1000:1000"
              chown -R 1000:1000 {{ .Values.persistence.mountPath }} || true
              chmod -R u+rwX,g+rwX {{ .Values.persistence.mountPath }} || true
          securityContext:
            runAsUser: 0
            runAsNonRoot: false
          volumeMounts:
            - name: data
              mountPath: {{ .Values.persistence.mountPath }}
          resources:
            requests:
              cpu: "50m"
              memory: "32Mi"
            limits:
              cpu: "200m"
              memory: "128Mi"
      {{- end }}
      containers:
        - name: hermes
          image: "{{ .Values.image.repository }}:{{ .Values.image.tag }}"
          imagePullPolicy: {{ .Values.image.pullPolicy }}
          {{- if .Values.securityContext }}
          securityContext:
            {{- toYaml .Values.securityContext | nindent 12 }}
          {{- end }}
          # NOTE: do NOT override command/args.
          # The image entrypoint (s6-overlay) starts the gateway itself; CLI flags like
          # --host/--port/--insecure are not recognized by the entrypoint.
          envFrom:
            - configMapRef:
                name: {{ include "hermes.fullname" . }}-config
            - secretRef:
                name: {{ include "hermes.fullname" . }}-env
          ports:
            - name: http
              containerPort: {{ .Values.hermes.dashboardPort }}
              protocol: TCP
          # ---- Probes ----
          # A startupProbe gates liveness/readiness so s6-overlay has plenty of time
          # to run the recursive chown of /opt/hermes during first boot.
          startupProbe:
            tcpSocket:
              port: http
            periodSeconds: {{ .Values.probes.startup.periodSeconds | default 10 }}
            failureThreshold: {{ .Values.probes.startup.failureThreshold | default 90 }}
            timeoutSeconds: {{ .Values.probes.startup.timeoutSeconds | default 3 }}
          readinessProbe:
            tcpSocket:
              port: http
            periodSeconds: {{ .Values.probes.readiness.periodSeconds | default 10 }}
            failureThreshold: {{ .Values.probes.readiness.failureThreshold | default 3 }}
            timeoutSeconds: {{ .Values.probes.readiness.timeoutSeconds | default 3 }}
          livenessProbe:
            tcpSocket:
              port: http
            periodSeconds: {{ .Values.probes.liveness.periodSeconds | default 30 }}
            failureThreshold: {{ .Values.probes.liveness.failureThreshold | default 5 }}
            timeoutSeconds: {{ .Values.probes.liveness.timeoutSeconds | default 3 }}
          resources:
            {{- toYaml .Values.resources | nindent 12 }}
          volumeMounts:
            - name: run
              mountPath: /run
            - name: tmp
              mountPath: /tmp
            {{- if .Values.persistence.enabled }}
            - name: data
              mountPath: {{ .Values.persistence.mountPath }}
            {{- end }}
            {{- with .Values.extraVolumeMounts }}
            {{- toYaml . | nindent 12 }}
            {{- end }}
      volumes:
        - name: run
          emptyDir:
            medium: Memory
            sizeLimit: 64Mi
        - name: tmp
          emptyDir:
            sizeLimit: 256Mi
        {{- if .Values.persistence.enabled }}
        - name: data
          persistentVolumeClaim:
            claimName: {{ include "hermes.fullname" . }}-data
        {{- end }}
        {{- with .Values.extraVolumes }}
        {{- toYaml . | nindent 8 }}
        {{- end }}
      {{- with .Values.nodeSelector }}
      nodeSelector:
        {{- toYaml . | nindent 8 }}
      {{- end }}
      {{- with .Values.affinity }}
      affinity:
        {{- toYaml . | nindent 8 }}
      {{- end }}
      {{- with .Values.tolerations }}
      tolerations:
        {{- toYaml . | nindent 8 }}
      {{- end }}
      terminationGracePeriodSeconds: {{ .Values.terminationGracePeriodSeconds | default 30 }}
